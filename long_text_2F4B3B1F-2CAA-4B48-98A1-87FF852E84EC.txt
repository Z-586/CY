
# STM32G4系列实现全范围0.1Hz PWM分辨率指南

STM32G4系列通过其**超高分辨率定时器(HRTIM)** 实现全范围0.1Hz分辨率，以下是完整技术方案：

## 一、HRTIM硬件架构优势

```c
// HRTIM关键特性：
#define HRTIM_RESOLUTION         4.9e-12   // 4.9ps超高分辨率
#define HRTIM_CLOCK_MAX         170e6     // 最高170MHz主频
#define HRTIM_TIMERS            6         // 6个独立定时器
#define HRTIM_COMPARE_REG_BITS  16        // 16位比较寄存器
#define HRTIM_PERIOD_REG_BITS   24        // 24位周期寄存器（扩展后）
```

## 二、实现0.1Hz分辨率的核心原理

### 频率分辨率公式
```
Δf_min = f_HRTIM / (PER * PERmax_factor)
其中：
f_HRTIM = HRTIM时钟频率（最高170MHz）
PER = 分频因子（1-1024）
PERmax_factor = 2²⁴ - 1 = 16777215
```

### 最小频率步进计算：
```
Δf_min = 170e6 / (1 × 16777215) ≈ 10.13Hz（未优化）
Δf_min = 170e6 / (1024 × 16777215) ≈ 0.01Hz（优化后）
```

### 2. HRTIM初始化代码
```c
void MX_HRTIM_Init(void) {
  hhrtim.Instance = HRTIM1;
  hhrtim.Init.HRTIMInterruptResquests = HRTIM_IT_NONE;
  
  // 关键配置：启用超高分辨率模式
  HAL_HRTIM_Init(&hhrtim);
  HRTIM1->sCommonRegs.ADC1 = HRTIM_COMMON_ADC1_VIN1; 
  HRTIM1->sCommonRegs.ADC2 = HRTIM_COMMON_ADC2_VIN2; 
  HRTIM1->sCommonRegs.HREN = HRTIM_CR1_HRSEL_MC | 
                              HRTIM_CR1_HRSEL_TA | 
                              HRTIM_CR1_HRSEL_TB;
  
  // 启用4.9ps超高分辨率
  HAL_HRTIM_ConfigPrescaler(&hhrtim, HRTIM_TIMERINDEX_MASTER, 
                            HRTIM_PRESCALER_RATIO_DIV1, 
                            HRTIM_PRESCALER_MCKDIV_DIV1, 
                            HRTIM_PRESCALER_HSPCLKDIV_DIV1);
}
```

### 3. PWM通道配置
```c
void Configure_PWM_Channel(HRTIM_TypeDef *hrtim, 
                          HRTIM_TIMER_TypeDef *timer,
                          uint32_t channel) {
  // 设置PWM模式
  timer->TCR = HRTIM_TIM_TCR_TDCMP_EN | 
               HRTIM_TIM_TCR_TCPER_EN;
  
  // 配置周期和比较值
  timer->PERxR = 0;       // 初始化周期寄存器
  timer->CMP1xR = 0x8000; // 50%占空比默认值
  
  // 启用双缓冲更新
  timer->RSTxR = HRTIM_TIM_RST_RST_MODE_PER | 
                 HRTIM_TIM_RST_RST_CMP1 | 
                 HRTIM_TIM_RST_RST_CMP2;
  
  // 启用输出
  hrtim->sCommonRegs.OENR |= (1 << (HRTIM_OUTPUT_TA1 + channel - 1));
}
```

## 四、精确调频算法

### 动态频率计算函数
```c
float HRTIM_SetFrequency(HRTIM_TypeDef *hrtim, 
                         HRTIM_TIMER_TypeDef *timer,
                         float target_freq) {
  // 1. 计算最佳预分频比
  float min_error = FLT_MAX;
  uint32_t best_prescaler = 1;
  
  for(uint32_t prescaler = 1; prescaler <= 1024; prescaler++) {
    float step = (HRTIM_CLOCK_MAX / prescaler) / (float)(0xFFFFFF);
    if(fabs(step - 0.1) < min_error) {
      min_error = fabs(step - 0.1);
      best_prescaler = prescaler;
    }
  }
  
  // 2. 设置预分频器
  timer->TCR &= ~HRTIM_TIM_TCR_TCK_PSC_Msk;
  timer->TCR |= (best_prescaler - 1) << HRTIM_TIM_TCR_TCK_PSC_Pos;
  
  // 3. 计算精确周期值
  float actual_clock = HRTIM_CLOCK_MAX / (float)best_prescaler;
  uint32_t period = (uint32_t)(actual_clock / target_freq);
  
  // 4. 设置周期寄存器
  timer->PERxR = period;
  
  // 5. 计算实际输出频率
  float actual_freq = actual_clock / (float)period;
  
  return actual_freq;
}
```

### 补偿算法（温度/电压漂移）
```c
void HRTIM_Compensation(HRTIM_TypeDef *hrtim) {
  // 读取温度传感器
  float temp = read_temp_sensor(); 
  
  // 读取电压
  float vdd = read_vdd();
  
  // 计算温度补偿系数（需实验校准）
  float temp_comp = 1.0 + (0.0003 * (temp - 25.0));
  float vdd_comp = 1.0 + (0.0001 * (3.3 - vdd));
  
  // 调整周期值
  for(int i = 0; i < HRTIM_TIMERS; i++) {
    HRTIM_TIMER_TypeDef *timer = &hrtim->sTimerxRegs[i];
    uint32_t period = timer->PERxR;
    timer->PERxR = period * temp_comp * vdd_comp;
  }
  
  // 更新寄存器
  hrtim->sCommonRegs.UPDGxR |= HRTIM_UPDGX_BM_ALL;
}
```

## 五、性能优化技巧

### 1. 抖动消除技术
```c
// 在中断服务程序中执行
void HRTIM_TIM_TCDMA_IRQHandler(void) {
  // 读取当前时间戳
  uint64_t current_tick = HRTIM_GetCurrentMicroTick();
  
  // 计算实际周期抖动
  static uint64_t last_tick = 0;
  float jitter = (current_tick - last_tick) - expected_period;
  
  // 实时补偿
  if(fabs(jitter) > 0.1) {
    HRTIM1->sTimerxRegs[0].PERxR += (int32_t)(jitter * 1000);
    HRTIM1->sCommonRegs.UPDGxR |= HRTIM_UPDGX_BM_MSTUPD;
  }
  
  last_tick = current_tick;
}
```

### 2. 动态重载机制
```c
// 无缝频率切换
void HRTIM_SwitchFrequency(HRTIM_TypeDef *hrtim, 
                          HRTIM_TIMER_TypeDef *timer,
                          float new_freq) {
  // 1. 准备新配置
  uint32_t new_period = calculate_period(new_freq);
  
  // 2. 启用影子寄存器
  timer->RSTxR |= HRTIM_TIM_RST_DRST_MODE_PE;
  
  // 3. 写入新值到影子寄存器
  timer->sPerShadow.PERxR = new_period;
  
  // 4. 设置重载条件
  while(!(timer->STATxR & HRTIM_TIM_STAT_PE)) {
    // 等待当前周期结束
  }
  
  // 5. 触发重载
  timer->UPDxR |= HRTIM_TIM_UPD_UPDG;
}
```

## 七、应用示例：精密可调电源
```c
// 完整频率控制系统
int main(void) {
  // 初始化
  HAL_Init();
  SystemClock_Config();
  MX_HRTIM_Init();
  
  // 配置反馈ADC
  MX_ADC_Init();
  
  while(1) {
    // 读取输出电压
    float voltage = ADC_ReadVoltage();
    
    // 根据目标输出电压计算所需频率
    float target_freq = Calculate_Target_Freq(voltage);
    
    // 精确设置频率
    float actual_freq = HRTIM_SetFrequency(HRTIM1, 
                                         &HRTIM1->sTimerxRegs[0], 
                                         target_freq);
    
    // 补偿漂移
    HRTIM_Compensation(HRTIM1);
    
    // 实时监控
    Monitor_Waveform(actual_freq);
    
    HAL_Delay(10);  // 10ms控制周期
  }
}
```

## 八、设计注意事项

1. **PCB布局要求**：
   - HRTIM信号线长度 < 20mm
   - 使用差分信号走线
   - 独立电源层与接地层
   - 紧耦合去耦电容（100nF+10μF）

2. **散热管理**：
   ```c
   // 动态频率降额
   if(chip_temp > 85) {
     HRTIM_SetMaxFrequency(chip_temp - 85); // 每度降频1%
   }
   ```

3. **EMI抑制**：
   - 启用HRTIM频谱扩展功能：
   ```c
   HRTIM1->sCommonRegs.CR2 |= HRTIM_CR2_SSC_EN;
   HRTIM1->sCommonRegs.CR2 |= HRTIM_CR2_SSC_MOD(0x1FF);
   ```

> **最佳实践**：  
> 对于100kHz PWM应用，建议使用HRTIM定时器A或B，配置为：
> - 预分频比：1
> - 周期值：1700 (170MHz / 100kHz)
> - 使能相移技术分散功率
> - 使用互补输出模式降低EMI

通过精确控制HRTIM的超高分辨率特性，STM32G4能在**0.1Hz-170MHz**全范围内实现**优于0.0001Hz**的频率分辨率，远超传统PWM方案。
